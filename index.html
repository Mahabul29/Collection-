require('dotenv').config(); const express = require('express'); const mongoose = require('mongoose'); const cors = require('cors'); const app = express(); const PORT = process.env.PORT || 5000; const FQDN = process.env.APP_FQDN || 'localhost'; const MONGO_URI = process.env.MONGO_URI || 'mongodb://localhost:27017/collection_db'; // Dynamic CORS using FQDN const allowedOrigins = [ process.env.FRONTEND_URL, `http://${FQDN}`, `https://${FQDN}`, 'http://localhost:3000' ]; app.use(cors({ origin: function (origin, callback) { if (!origin || allowedOrigins.includes(origin)) { callback(null, true); } else { callback(new Error('CORS policy violation for this FQDN')); } }, credentials: true })); app.use(express.json()); // MongoDB Connection mongoose.connect(MONGO_URI) .then(() => console.log('MongoDB Connected Successfully')) .catch(err => console.error('MongoDB Error:', err)); app.listen(PORT, () => { console.log(`Server running on FQDN: http://${FQDN}:${PORT}`); });
